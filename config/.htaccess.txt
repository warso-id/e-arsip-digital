# ============================================
# 1. SECURITY HEADERS (CRITICAL)
# ============================================
<IfModule mod_headers.c>
    # --------------------------------------------------
    # Content Security Policy (CSP)
    # --------------------------------------------------
    # KEAMANAN KRITIS: Mencegah XSS, injection, data exfiltration
    # Whitelist CDN domains yang digunakan aplikasi
    Header always set Content-Security-Policy "\
        default-src 'self'; \
        script-src 'self' 'unsafe-inline' 'unsafe-eval' \
            https://apis.google.com \
            https://cdn.jsdelivr.net \
            https://cdnjs.cloudflare.com \
            https://unpkg.com \
            https://script.google.com \
            https://script.googleusercontent.com; \
        style-src 'self' 'unsafe-inline' \
            https://cdn.jsdelivr.net \
            https://cdnjs.cloudflare.com \
            https://unpkg.com; \
        img-src 'self' data: blob: https:; \
        font-src 'self' \
            https://cdn.jsdelivr.net \
            https://cdnjs.cloudflare.com; \
        connect-src 'self' \
            https://script.google.com \
            https://script.googleusercontent.com \
            https://cdn.jsdelivr.net \
            https://cdnjs.cloudflare.com; \
        frame-src 'self' https://script.google.com; \
        form-action 'self' https://script.google.com; \
        base-uri 'self'; \
        manifest-src 'self'; \
        worker-src 'self' blob:; \
        object-src 'none'; \
        upgrade-insecure-requests; \
    "

    # --------------------------------------------------
    # Anti-MIME Sniffing
    # --------------------------------------------------
    Header always set X-Content-Type-Options "nosniff"

    # --------------------------------------------------
    # Anti-Clickjacking (PWA COMPATIBLE)
    # --------------------------------------------------
    # NOTE: DENY akan merusak PWA standalone mode!
    # Gunakan SAMEORIGIN untuk PWA compatibility
    Header always set X-Frame-Options "SAMEORIGIN"

    # --------------------------------------------------
    # XSS Protection (Legacy browsers)
    # --------------------------------------------------
    Header always set X-XSS-Protection "1; mode=block"

    # --------------------------------------------------
    # Referrer Policy
    # --------------------------------------------------
    Header always set Referrer-Policy "strict-origin-when-cross-origin"

    # --------------------------------------------------
    # Permissions Policy
    # --------------------------------------------------
    Header always set Permissions-Policy "\
        accelerometer=(), \
        ambient-light-sensor=(), \
        autoplay=(), \
        battery=(), \
        camera=(), \
        display-capture=(), \
        fullscreen=(self), \
        geolocation=(), \
        gyroscope=(), \
        interest-cohort=(), \
        magnetometer=(), \
        microphone=(), \
        midi=(), \
        payment=(), \
        picture-in-picture=(), \
        screen-wake-lock=(), \
        sync-xhr=(), \
        usb=(), \
        web-share=() \
    "

    # --------------------------------------------------
    # Cross-Origin Policies (CDN COMPATIBLE)
    # --------------------------------------------------
    # COOP: Allow popups (needed for OAuth, print windows)
    Header always set Cross-Origin-Opener-Policy "same-origin-allow-popups"
    
    # CORP: cross-origin agar CDN resources tetap load
    # JANGAN gunakan 'same-origin' - akan memblokir jsDelivr/Cloudflare!
    Header always set Cross-Origin-Resource-Policy "cross-origin"
    
    # COEP: JANGAN gunakan 'require-corp' - akan merusak semua resource CDN!
    # Hanya aktifkan jika SEMUA resource memiliki header CORP yang sesuai
    # Header always set Cross-Origin-Embedder-Policy "unsafe-none"

    # --------------------------------------------------
    # Remove Server Info Headers
    # --------------------------------------------------
    Header always unset X-Powered-By
    # Header always unset Server  # Tidak selalu berfungsi, gunakan di httpd.conf: ServerTokens Prod

    # --------------------------------------------------
    # Cache Control untuk HTML (Dynamic Pages)
    # --------------------------------------------------
    <FilesMatch "\.(html|htm|php)$">
        Header always set Cache-Control "no-store, no-cache, must-revalidate, private"
        Header always set Pragma "no-cache"
        Header always set Expires "0"
    </FilesMatch>

    # --------------------------------------------------
    # Cache Control untuk Static Assets (Immutable)
    # --------------------------------------------------
    <FilesMatch "\.(css|js|mjs)$">
        Header always set Cache-Control "public, max-age=31536000, immutable"
    </FilesMatch>

    # --------------------------------------------------
    # Cache Control untuk Fonts
    # --------------------------------------------------
    <FilesMatch "\.(woff|woff2|ttf|otf|eot)$">
        Header always set Cache-Control "public, max-age=31536000, immutable"
    </FilesMatch>

    # --------------------------------------------------
    # Cache Control untuk Images
    # --------------------------------------------------
    <FilesMatch "\.(jpg|jpeg|png|gif|svg|webp|ico)$">
        Header always set Cache-Control "public, max-age=2592000"
    </FilesMatch>

    # --------------------------------------------------
    # PWA: Manifest (Short Cache)
    # --------------------------------------------------
    <FilesMatch "^manifest\.(json|webmanifest)$">
        Header always set Cache-Control "public, max-age=604800"
        Header always set Content-Type "application/manifest+json"
    </FilesMatch>

    # --------------------------------------------------
    # PWA: Service Worker (Very Short Cache)
    # --------------------------------------------------
    <FilesMatch "^sw\.js$">
        Header always set Cache-Control "public, max-age=86400"
        Header always set Service-Worker-Allowed "/"
    </FilesMatch>
</IfModule>

# --------------------------------------------------
# Protect Configuration & Sensitive Files
# --------------------------------------------------

</FilesMatch>

    </IfModule>
    <IfModule !mod_authz_core.c>

</FilesMatch>

# 3. URL REWRITING & ROUTING
# ============================================
<IfModule mod_rewrite.c>
    RewriteEngine On
    
    # --------------------------------------------------
    # Force HTTPS (Uncomment in production)
    # --------------------------------------------------
    # RewriteCond %{HTTPS} off
    # RewriteCond %{HTTP:X-Forwarded-Proto} !https
    # RewriteRule ^(.*)$ https://%{HTTP_HOST}/$1 [R=301,L]
    
    # --------------------------------------------------
    # Force WWW (atau non-WWW, pilih salah satu)
    # --------------------------------------------------
    # RewriteCond %{HTTP_HOST} !^www\. [NC]
    # RewriteRule ^(.*)$ https://www.%{HTTP_HOST}/$1 [R=301,L]
    
    # --------------------------------------------------
    # Remove Trailing Slash (kecuali direktori)
    # --------------------------------------------------
    RewriteCond %{REQUEST_FILENAME} !-d
    RewriteRule ^(.+)/$ /$1 [R=301,L]
    
    # --------------------------------------------------
    # SPA Routing - Fallback ke index.html
    # --------------------------------------------------
    # Jika file/folder tidak ada DAN bukan static asset,
    # redirect ke index.html untuk client-side routing
    RewriteCond %{REQUEST_FILENAME} !-f
    RewriteCond %{REQUEST_FILENAME} !-d
    RewriteCond %{REQUEST_URI} !\.(css|js|mjs|png|jpg|jpeg|gif|svg|webp|ico|woff|woff2|ttf|otf|eot|pdf|doc|docx|xls|xlsx|json|xml|txt|map|webmanifest)$
    RewriteRule ^(.*)$ index.html [L,QSA]
    
    # --------------------------------------------------
    # Redirect Old/Deprecated URLs
    # --------------------------------------------------
    # Redirect old dashboard paths to new ones
    RewriteRule ^dashboard/?$ dashboard/index.html [L,R=301]
    
    # --------------------------------------------------
    # Block SQL Injection Attempts
    # --------------------------------------------------
    RewriteCond %{QUERY_STRING} (\<|%3C).*script.*(\>|%3E) [NC,OR]
    RewriteCond %{QUERY_STRING} GLOBALS(=|\[|\%[0-9A-Z]{0,2}) [NC,OR]
    RewriteCond %{QUERY_STRING} _REQUEST(=|\[|\%[0-9A-Z]{0,2}) [NC,OR]
    RewriteCond %{QUERY_STRING} (SELECT|INSERT|UPDATE|DELETE|UNION|DROP|ALTER|TRUNCATE|EXEC|DECLARE|CREATE) [NC,OR]
    RewriteCond %{QUERY_STRING} (\.\./|\.\.\\|%2e%2e|%2e%2e/) [NC,OR]
    RewriteCond %{QUERY_STRING} (etc/passwd|/etc/shadow|/proc/self) [NC,OR]
    RewriteCond %{QUERY_STRING} (<|>|'|%3C|%3E|%27) [NC]
    RewriteRule ^(.*)$ - [F,L]
    
    # --------------------------------------------------
    # Block Directory Traversal
    # --------------------------------------------------
    RewriteCond %{REQUEST_URI} (\.\./|\.\.\\|%2e%2e|%2e%2e/) [NC,OR]
    RewriteCond %{REQUEST_URI} (etc/passwd|cmd\.exe|cmd\.com) [NC]
    RewriteRule ^(.*)$ - [F,L]
    
    # --------------------------------------------------
    # Block Access to Sensitive Files via URL
    # --------------------------------------------------
    RewriteRule ^(config\.js|\.env|\.git|composer\.json|package\.json|Dockerfile|docker-compose\.yml) - [F,L]
    
    # --------------------------------------------------
    # Block Bad Bots & Scanners (Optional)
    # --------------------------------------------------
    # RewriteCond %{HTTP_USER_AGENT} (bot|crawl|scan|spider|scrape|sqlmap|nikto|acunetix|nessus) [NC]
    # RewriteRule ^(.*)$ - [F,L]
</IfModule>

# ============================================
# 4. COMPRESSION (GZIP/DEFLATE)
# ============================================
<IfModule mod_deflate.c>
    # --------------------------------------------------
    # Compress Text-based Content
    # --------------------------------------------------
    AddOutputFilterByType DEFLATE text/html
    AddOutputFilterByType DEFLATE text/plain
    AddOutputFilterByType DEFLATE text/xml
    AddOutputFilterByType DEFLATE text/css
    AddOutputFilterByType DEFLATE text/javascript
    AddOutputFilterByType DEFLATE text/csv
    AddOutputFilterByType DEFLATE text/markdown
    
    # --------------------------------------------------
    # Compress Application Content
    # --------------------------------------------------
    AddOutputFilterByType DEFLATE application/javascript
    AddOutputFilterByType DEFLATE application/json
    AddOutputFilterByType DEFLATE application/xml
    AddOutputFilterByType DEFLATE application/rss+xml
    AddOutputFilterByType DEFLATE application/atom+xml
    AddOutputFilterByType DEFLATE application/x-javascript
    AddOutputFilterByType DEFLATE application/manifest+json
    AddOutputFilterByType DEFLATE application/x-web-app-manifest+json
    
    # --------------------------------------------------
    # Compress Fonts & SVG
    # --------------------------------------------------
    AddOutputFilterByType DEFLATE image/svg+xml
    AddOutputFilterByType DEFLATE application/vnd.ms-fontobject
    AddOutputFilterByType DEFLATE application/x-font-ttf
    AddOutputFilterByType DEFLATE font/opentype
    AddOutputFilterByType DEFLATE font/ttf
    
    # --------------------------------------------------
    # Remove Browser Bugs (Legacy)
    # --------------------------------------------------
    BrowserMatch ^Mozilla/4 gzip-only-text/html
    BrowserMatch ^Mozilla/4\.0[678] no-gzip
    BrowserMatch \bMSIE !no-gzip !gzip-only-text/html
    
    # --------------------------------------------------
    # Vary Header for Proxy Caches
    # --------------------------------------------------
    Header append Vary Accept-Encoding env=!dont-vary
</IfModule>

# ============================================
# 6. EXPIRES / CACHE (Fallback)
# ============================================
<IfModule mod_expires.c>
    ExpiresActive On
    
    # Default: 1 month
    ExpiresDefault "access plus 1 month"
    
    # HTML: No cache
    ExpiresByType text/html "access plus 0 seconds"
    
    # CSS, JS: 1 year (immutable via Cache-Control header)
    ExpiresByType text/css "access plus 1 year"
    ExpiresByType application/javascript "access plus 1 year"
    
    # Images: 1 month
    ExpiresByType image/jpeg "access plus 1 month"
    ExpiresByType image/png "access plus 1 month"
    ExpiresByType image/gif "access plus 1 month"
    ExpiresByType image/svg+xml "access plus 1 month"
    ExpiresByType image/webp "access plus 1 month"
    ExpiresByType image/x-icon "access plus 1 year"
    
    # Fonts: 1 year
    ExpiresByType application/font-woff "access plus 1 year"
    ExpiresByType application/font-woff2 "access plus 1 year"
    ExpiresByType application/x-font-ttf "access plus 1 year"
    ExpiresByType font/opentype "access plus 1 year"
    
    # PWA Manifest: 1 week
    ExpiresByType application/manifest+json "access plus 1 week"
</IfModule>
